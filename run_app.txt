@echo off
setlocal
title IOP Benthic Survey Prep Tool v0.2.0
cd /d "%~dp0"
set PYTHONUTF8=1
set PYTHONIOENCODING=utf-8

echo ====================================================================
echo  IOP Benthic Survey Prep Tool - v0.2.0 - Standalone Desktop Mode
echo ====================================================================
echo.

set "PY="
where py >nul 2>&1 && set "PY=py -3"
if not defined PY where python >nul 2>&1 && set "PY=python"

if not defined PY goto nopython
%PY% --version >nul 2>&1
if errorlevel 1 goto nopython

%PY% -c "import fastapi, uvicorn, cv2, numpy, pydantic" >nul 2>&1
if errorlevel 1 goto nodeps

if not exist "frontend\dist\index.html" (
    echo [WARNING] frontend\dist\index.html not found. Run "npm run build" in the frontend folder first.
    echo.
)

echo Starting server and opening application...
echo You can also open your browser manually at: http://127.0.0.1:8000
echo.
%PY% server.py
echo.
echo [NOTICE] Server stopped. Exit code: %errorlevel%
goto end

:nopython
echo [ERROR] Python 3.10+ was not found. Install it and tick "Add Python to PATH".
goto end

:nodeps
echo [ERROR] Missing Python packages. Run:
echo     pip install fastapi uvicorn opencv-python numpy pydantic
goto end

:end
echo.
pause
endlocal
